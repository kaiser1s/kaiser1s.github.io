MyPlugin.java (Plugin Example)

package com.dev;
import org.bukkit.Bukkit;
import org.bukkit.command.Command;
import org.bukkit.command.CommandExecutor;
import org.bukkit.command.CommandSender;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.player.PlayerJoinEvent;
import org.bukkit.plugin.java.JavaPlugin;

public class MyPlugin extends JavaPlugin implements Listener, CommandExecutor {
    @Override
    public void onEnable() {
        Bukkit.getPluginManager().registerEvents(this, this);
        getCommand("hello").setExecutor(this);
        getLogger().info("MyPlugin enabled!");
    }

    @EventHandler
    public void onJoin(PlayerJoinEvent event) {
        event.getPlayer().sendMessage("§aWelcome! The plugin is active.");
    }

    @Override
    public boolean onCommand(CommandSender sender, Command command, String label, String[] args) {
        if (sender instanceof Player) {
            Player player = (Player) sender;
            player.sendMessage("§bHello " + player.getName() + "!");
        } else {
            sender.sendMessage("Hello console!");
        }
        return true;
    }
}

AdminPanel.sk (Skript Example)

command /adminpanel:
    permission: op
    permission message: &cYou do not have permission to use the Admin Panel!
    trigger:
        set {_gui} to a new chest inventory with 2 rows named "&8&lAdmin Control Panel"
        set slot 0 of {_gui} to red wool named "&4&lClose Server" with lore "&7Restarts the server immediately."
        set slot 1 of {_gui} to white wool named "&e&lSkripts Reload" with lore "&7Runs /sk reload scripts"
        set slot 10 of {_gui} to light blue wool named "&b&lCreative Mode"
        set slot 11 of {_gui} to green wool named "&2&lSurvival Mode"
        open {_gui} to player

on inventory click:
    if name of event-inventory is "&8&lAdmin Control Panel":
        cancel event
        if index of event-slot is 0:
            execute console command "restart"
        if index of event-slot is 1:
            execute player command "sk reload scripts"
        if index of event-slot is 10:
            set player's gamemode to creative
        if index of event-slot is 11:
            set player's gamemode to survival
